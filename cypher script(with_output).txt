// 1. Indexes
CREATE INDEX concept_sctid IF NOT EXISTS FOR (c:Concept) ON (c.sctid);
CREATE INDEX term_id IF NOT EXISTS FOR (t:Term) ON (t.termId);
CREATE FULLTEXT INDEX term_search IF NOT EXISTS
  FOR (t:Term) ON EACH [t.term];

//Output:



// 2. Concepts
LOAD CSV WITH HEADERS FROM 'file:///nodes_concepts.csv' AS row
CREATE (:Concept {sctid: row.sctid, conceptType: row.conceptType,
                  definitionStatus: row.definitionStatus});

//Output:
Created 10,235 nodes, set 30,705 properties, added 10,235 labels


// 3. Terms
LOAD CSV WITH HEADERS FROM 'file:///nodes_terms.csv' AS row
CREATE (:Term {termId: row.termId, term: row.term});

//Output:
Created 29,180 nodes, set 58,360 properties, added 29,180 labels


// 4. HAS_TERM
LOAD CSV WITH HEADERS FROM 'file:///rels_has_term.csv' AS row
MATCH (c:Concept {sctid: row.startId})
MATCH (t:Term {termId: row.endId})
CREATE (c)-[:HAS_TERM {termType: row.termType}]->(t);

//Output:
Created 29,180 relationships, set 29,180 properties


// 5. SNOMED rels (needs APOC)

MATCH (target:Concept {sctid: row.endId})

YIELD rel RETURN count(rel);

// Test 16 — FULL END-TO-END NSTEMI REASONING CHAIN
MATCH path = (chest:Concept {sctid:'29857009'})
  -[:HAS_DIFFERENTIAL]->(acs:Concept {sctid:'394659003'})
  -[:REQUIRES_EXAM]->(troponin:Concept {sctid:'105000003'})
  -[:ALGORITHM_RESULT {outcome:'rule_in'}]->(conclusion:DiagnosticConclusion)
  -[:CONFIRMS]->(nstemi:Concept {sctid:'401314000'})
  -[:REQUIRES_EXAM {urgency:'within_24h'}]->(angio:Concept {sctid:'33367005'})
WITH nodes(path) AS chain
UNWIND chain AS n
OPTIONAL MATCH (n)-[:HAS_TERM]->(t:Term)
WHERE t.termType = 'FSN'
WITH n, t
WITH n, collect(t.term)[0] AS term_name
WITH collect(coalesce(n.fsn, term_name)) AS node_names
RETURN node_names AS reasoning_chain, size(node_names) AS total_nodes;
// Output: 1 row, 6 nodes in correct order
